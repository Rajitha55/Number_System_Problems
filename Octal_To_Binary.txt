//Program to convert Octal to Binary
//Here, we are using stoi() first to convrt octal->decimal
//Then, we will use division method to convert decimal-> binary

#include <bits/stdc++.h>
using namespace std;

int main()
{
    string octal;
    cout<<"Enter Octal number:"<<endl;
    cin>>octal;
    int decimal=stoi(octal,0,8); //octal->decimal
    string bin="";
    while(decimal>0)  //decimal-> binary
    {
        bin=char((decimal%2)+'0') + bin;
        decimal/=2;
    }
    
    cout<<bin;
    
    return 0;
}
